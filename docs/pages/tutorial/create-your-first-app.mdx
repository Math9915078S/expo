// Agenda Alfa - App de Estética Automotiva
// Criado com React Native + Expo

import React, { useState, useEffect } from 'react';
import { View, Text, TextInput, Button, FlatList, TouchableOpacity, Linking, StyleSheet } from 'react-native';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { NavigationContainer } from '@react-navigation/native';
import { createNativeStackNavigator } from '@react-navigation/native-stack';

const Stack = createNativeStackNavigator();

const STORAGE_KEY = '@agenda_alfa_agendamentos';

function HomeScreen({ navigation }) {
  const [agendamentos, setAgendamentos] = useState([]);

  useEffect(() => {
    const loadData = async () => {
      const data = await AsyncStorage.getItem(STORAGE_KEY);
      if (data) setAgendamentos(JSON.parse(data));
    };
    const unsubscribe = navigation.addListener('focus', loadData);
    return unsubscribe;
  }, [navigation]);

  const sendWhatsApp = (item) => {
    const msg = `Olá, ${item.nome}! Seu agendamento com a Alfa Estética Automotiva está confirmado para dia ${item.data} às ${item.hora}. Serviço: ${item.servico}. Valor: R$${item.valor}.`;
    const url = `https://wa.me/${item.telefone}?text=${encodeURIComponent(msg)}`;
    Linking.openURL(url);
  };

  return (
    <View style={styles.container}>
      <Button title="Novo Agendamento" onPress={() => navigation.navigate('Novo')} />
      <FlatList
        data={agendamentos}
        keyExtractor={(_, i) => i.toString()}
        renderItem={({ item }) => (
          <View style={styles.card}>
            <Text style={styles.nome}>{item.nome}</Text>
            <Text>{item.data} às {item.hora}</Text>
            <Text>{item.veiculo} - {item.servico}</Text>
            <Text>R$ {item.valor}</Text>
            <TouchableOpacity style={styles.whatsappBtn} onPress={() => sendWhatsApp(item)}>
              <Text style={styles.whatsappText}>Confirmar via WhatsApp</Text>
            </TouchableOpacity>
          </View>
        )}
      />
    </View>
  );
}

function NovoAgendamento({ navigation }) {
  const [nome, setNome] = useState('');
  const [telefone, setTelefone] = useState('');
  const [data, setData] = useState('');
  const [hora, setHora] = useState('');
  const [veiculo, setVeiculo] = useState('');
  const [servico, setServico] = useState('');
  const [valor, setValor] = useState('');

  const salvar = async () => {
    const novo = { nome, telefone, data, hora, veiculo, servico, valor };
    const antigos = await AsyncStorage.getItem(STORAGE_KEY);
    const lista = antigos ? JSON.parse(antigos) : [];
    lista.push(novo);
    await AsyncStorage.setItem(STORAGE_KEY, JSON.stringify(lista));
    navigation.goBack();
  };

  return (
    <View style={styles.container}>
      <TextInput placeholder="Nome" style={styles.input} onChangeText={setNome} />
      <TextInput placeholder="Telefone (somente números)" style={styles.input} keyboardType="phone-pad" onChangeText={setTelefone} />
      <TextInput placeholder="Data (dd/mm/aaaa)" style={styles.input} onChangeText={setData} />
      <TextInput placeholder="Hora (hh:mm)" style={styles.input} onChangeText={setHora} />
      <TextInput placeholder="Veículo" style={styles.input} onChangeText={setVeiculo} />
      <TextInput placeholder="Serviço" style={styles.input} onChangeText={setServico} />
      <TextInput placeholder="Valor (R$)" style={styles.input} keyboardType="numeric" onChangeText={setValor} />
      <Button title="Salvar Agendamento" onPress={salvar} />
    </View>
  );
}

export default function App() {
  return (
    <NavigationContainer>
      <Stack.Navigator>
        <Stack.Screen name="Agenda Alfa" component={HomeScreen} />
        <Stack.Screen name="Novo" component={NovoAgendamento} />
      </Stack.Navigator>
    </NavigationContainer>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, padding: 16 },
  input: { borderWidth: 1, marginBottom: 8, padding: 8, borderRadius: 6 },
  card: { padding: 12, marginBottom: 12, borderWidth: 1, borderRadius: 6 },
  nome: { fontWeight: 'bold', fontSize: 16 },
  whatsappBtn: { marginTop: 8, backgroundColor: '#25D366', padding: 8, borderRadius: 6 },
  whatsappText: { color: '#fff', textAlign: 'center' }
});
